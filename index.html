const esc = s => String(s ?? '').replace(/[&<>"']/g, c => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c]));

const g = (r, k) => String(r[k] ?? '').trim();

function num(v) {

  if (v === '' || v == null) return null;

  const n = parseFloat(String(v).replace(/[^0-9.\-]/g, ''));

  return Number.isFinite(n) ? n : null;

}

function oddsNum(v) { const n = num(v); return n != null && Math.abs(n) >= 100 ? n : null; }

const signed = n => (n > 0 ? '+' + n : String(n));

const money = n => (Math.abs(n) < 0.005 ? '' : n < 0 ? '−' : '+') + Math.abs(n).toFixed(2) + 'u';

const cls = n => (n > 0.005 ? 'pos' : n < -0.005 ? 'neg' : '');

const color = n => COLORS[n] || '#8C9098';

// Sheets turns 3PM into 3:00 PM and 3PA into 3:00 AM; undo that.

function fixStat(s) {

  if (/^3:00(:00)?\s*PM$/i.test(s)) return '3PM';

  if (/^3:00(:00)?\s*AM$/i.test(s)) return '3PA';

  return s;

}


function parseCSV(t) {

  const rows = []; let row = [], f = '', q = false;

  for (let i = 0; i < t.length; i++) {

    const c = t[i];

    if (q) { if (c === '"') { if (t[i + 1] === '"') { f += '"'; i++; } else q = false; } else f += c; }

    else if (c === '"') q = true;

    else if (c === ',') { row.push(f); f = ''; }

    else if (c === '\n') { row.push(f); rows.push(row); row = []; f = ''; }

    else if (c !== '\r') f += c;

  }

  if (f !== '' || row.length) { row.push(f); rows.push(row); }

  return rows;

}

function toObjects(rows) {

  if (!rows.length) return [];

  const h = rows[0].map(x => x.trim());

  return rows.slice(1).filter(r => r.some(c => c.trim() !== '')).map(r => Object.fromEntries(h.map((k, i) => [k, r[i] ?? ''])));

}


function parseGame(game, ts) {

  const m = game.match(/^\w{3}, (\w{3}) (\d{1,2}), (\d{4}) (v|@) (.+)$/);

  if (m) {

    const team = m[5];

    const nick = /Trail Blazers$/.test(team) ? 'Blazers' : team.split(' ').pop();

    return { date: new Date(+m[3], MONTHS[m[1]], +m[2]), short: `${MONTHS[m[1]] + 1}/${m[2]}`,

      title: `${m[4] === 'v' ? 'vs.' : 'at'} ${team}`, opp: nick, home: m[4] === 'v' };

  }

  const d = new Date(ts);

  return { date: isNaN(d) ? new Date(0) : d, short: game.replace(/ Game$/, '').replace('NBA Cup ', 'Cup '), title: game, opp: 'Opponent', home: true };

}


/* ---------- describe a pick in plain words ---------- */

function describe(seg, f, ctx) {

  const per = f.period && f.period !== 'Full game' ? f.period + ' ' : '';

  const ou = f.ou && f.ou !== 'None' ? f.ou : '';

  const line = f.line;

  const team = s => (s === 'Knicks' ? 'Knicks' : s === 'Opponent' ? ctx.opp : s === 'Game' ? `Knicks/${ctx.opp}` : s);

  let text;

  if (f.bet === 'Other') text = f.details || 'Other';

  else if (f.bet === 'Spread') text = `${team(f.subject)} ${line == null ? '' : signed(line)}`;

  else if (f.bet === 'Moneyline') text = `${team(f.subject)} ML`;

  else if (f.bet === 'Total') text = f.subject === 'Game' || !f.subject ? `${team('Game')} ${ou} ${line ?? ''}` : `${team(f.subject)} TT ${ou} ${line ?? ''}`;

  else if (f.bet === 'Team total') text = `${team(f.subject)} TT ${ou} ${line ?? ''}`;

  else if (/^Team /.test(f.bet)) text = `${team(f.subject)} ${ou} ${line ?? ''} ${f.bet.replace('Team ', '').replace('3PM', 'threes')}`;

  else {

    const who = f.player || f.subject;

    text = ou === 'Yes' || line == null ? `${who} ${f.bet.toLowerCase()}` : `${who} ${ou} ${line} ${f.bet}`;

  }

  text = (per + text).replace(/\s+/g, ' ').trim();

  if (f.bet !== 'Other' && f.details) text += ` (${f.details})`;

  return text;

}


function readPick(r, seg, ctx) {

  const k = seg.key;

  const v = name => g(r, k + ': ' + name);

  let f;

  if (k === 'Prop') {

    const p = v('Player');

    f = { subject: p, player: p === 'Opponent player' ? v('Opponent player name') || 'Opponent player' : p,

      bet: fixStat(v('Stat')) };

  } else {

    const s = v('Subject');

    f = { subject: s, player: s === 'Opponent player' ? v('Opponent player name') || 'Opponent player' : s,

      bet: fixStat(v('Bet')), details: v('Details') };

  }

  f.ou = v('Over/Under'); f.line = num(v('Line')); f.period = v('Period');

  const odds = oddsNum(v('Odds')), boost = oddsNum(v('Boosted odds'));

  const eff = boost ?? odds;

  const raw = g(r, 'Result: ' + k).toLowerCase();

  const result = ['win', 'loss', 'push', 'void'].includes(raw) ? raw : 'pending';

  const profit = result === 'loss' ? -STAKE : result === 'win' && eff != null ? (eff > 0 ? STAKE * eff / 100 : STAKE * 100 / Math.abs(eff)) : 0;

  const empty = !f.bet && !f.subject;

  return { seg, text: empty ? '' : describe(seg, f, ctx), odds, boost, eff, result: empty ? 'none' : result, profit };

}


function build(rows) {

  const byShow = new Map();

  rows.forEach(r => {

    const game = g(r, 'Game'), name = g(r, 'Name');

    if (!game || !name) return;

    if (!byShow.has(game)) byShow.set(game, { game, info: parseGame(game, g(r, 'Timestamp')), hosts: new Map() });

    const show = byShow.get(game);

    const ctx = { opp: show.info.opp };

    show.hosts.set(name, SEGMENTS.map(seg => readPick(r, seg, ctx)));  // latest row for a host wins

  });

  const shows = [...byShow.values()].sort((a, b) => a.info.date - b.info.date);

  const picks = [];

  shows.forEach((s, i) => s.hosts.forEach((ps, host) => ps.forEach(p => { if (p.result !== 'none') picks.push({ ...p, host, showIndex: i }); })));

  return { shows, picks };

}


function hostsList() {

  const seen = [...new Set(state.picks.map(p => p.host))];

  return HOST_ORDER.filter(h => seen.includes(h)).concat(seen.filter(h => !HOST_ORDER.includes(h)));

}


function tally(list) {

  const t = { w: 0, l: 0, p: 0, open: 0, profit: 0 };

  list.forEach(x => {

    if (x.result === 'win') t.w++; else if (x.result === 'loss') t.l++;

    else if (x.result === 'push' || x.result === 'void') t.p++; else t.open++;

    t.profit += x.profit;

  });

  t.graded = t.w + t.l;

  return t;

}

const rec = t => `${t.w}-${t.l}${t.p ? '-' + t.p : ''}`;

const pct = t => (t.graded ? Math.round(t.w / t.graded * 100) + '%' : '–');

      <td class="num ${cls(r.t.profit)}"><strong>${money(r.t.profit)}</strong></td>

      <td class="num ${cls(r.t.profit)}">${r.t.graded ? Math.round(r.t.profit / ((r.t.graded + r.t.p) * STAKE) * 100) + '%' : '–'}</td></tr>`).join('') +

    `<tr><td><strong>The show</strong></td><td class="num">${state.shows.length}</td><td class="num">${rec(all)}</td><td class="num">${pct(all)}</td><td class="num">${all.open}</td>

      <td class="num ${cls(all.profit)}"><strong>${money(all.profit)}</strong></td><td class="num ${cls(all.profit)}">${all.graded ? Math.round(all.profit / ((all.graded + all.p) * STAKE) * 100) + '%' : '–'}</td></tr></tbody>`;

}

function renderChart() {

  const hosts = hostsList();

  const shows = state.shows;

  if (!state.picks.some(p => p.result !== 'pending')) {

    $('chart').innerHTML = '<div class="empty">The chart starts once the first pick is graded.</div>'; $('legend').innerHTML = ''; return;

  }

  const W = 820, H = 300, L = 58, R = 18, T = 14, B = 34;

  const series = Object.fromEntries(hosts.map(h => [h, [0]]));

  shows.forEach((s, i) => hosts.forEach(h => {

    const add = state.picks.filter(p => p.host === h && p.showIndex === i).reduce((a, p) => a + p.profit, 0);

    series[h].push(series[h].at(-1) + add);

  }));

  const vals = hosts.flatMap(h => series[h]);

  let lo = Math.min(0, ...vals), hi = Math.max(0, ...vals);

  const step = Math.max(0.5, Math.ceil((hi - lo) / 5 / 0.5) * 0.5);

  lo = Math.floor(lo / step) * step; hi = Math.ceil(hi / step) * step || step;

  const x = i => L + (W - L - R) * (i / Math.max(shows.length, 1));

  const y = v => T + (H - T - B) * (1 - (v - lo) / (hi - lo));

  let svg = `<svg viewBox="0 0 ${W} ${H}" role="img" aria-label="Running units by host over the season">`;

  for (let v = lo; v <= hi + 0.1; v += step) {

    svg += `<line x1="${L}" x2="${W - R}" y1="${y(v)}" y2="${y(v)}" style="stroke:var(--rule);stroke-width:${v === 0 ? 2 : 1}"/>` +

      `<text x="${L - 8}" y="${y(v) + 4}" text-anchor="end" style="fill:var(--mute);font:12px var(--body)">${Math.abs(v) < 0.001 ? '0u' : money(v)}</text>`;

  }

  const every = Math.ceil(shows.length / 8);

  shows.forEach((s, i) => { if (i % every === 0 || i === shows.length - 1)

    svg += `<text x="${x(i + 1)}" y="${H - 10}" text-anchor="middle" style="fill:var(--mute);font:12px var(--body)">${esc(s.info.short)}</text>`; });

  hosts.forEach(h => {

    const pts = series[h].map((v, i) => `${x(i).toFixed(1)},${y(v).toFixed(1)}`).join(' ');

    svg += `<polyline points="${pts}" style="fill:none;stroke:${color(h)};stroke-width:3;stroke-linejoin:round"/>`;

    const last = series[h].length - 1;

    svg += `<circle cx="${x(last)}" cy="${y(series[h][last])}" r="4.5" style="fill:${color(h)}"/>`;

  });

  $('chart').innerHTML = svg + '</svg>';

  $('legend').innerHTML = hosts.map(h => `<span><span class="dot" style="background:${color(h)}"></span>${esc(h)} <span class="${cls(series[h].at(-1))}">${money(series[h].at(-1))}</span></span>`).join('');

}

const LABEL = { win: 'Win', loss: 'Loss', push: 'Push', void: 'Void', pending: 'Pending' };


function renderLog() {

  const list = state.shows.filter(s => state.host === 'All' || s.hosts.has(state.host)).slice().reverse();

  if (!list.length) { $('log').innerHTML = '<div class="empty">No shows for this host yet.</div>'; return; }

  const hostOrder = hostsList();

  $('log').innerHTML = list.map(s => {

    const names = hostOrder.filter(h => s.hosts.has(h));

    const nights = names.map(h => {

      const ps = s.hosts.get(h).filter(p => p.result !== 'none');

      return { h, ps, t: tally(ps) };

    });

    const graded = nights.every(n => n.t.open === 0) && nights.length > 1;

    let verdict = '';

    if (graded) {

      const best = nights.slice().sort((a, b) => b.t.profit - a.t.profit);

      verdict = Math.abs(best[0].t.profit - best[1].t.profit) < 0.005

        ? 'Even night'

        : `<b>${esc(best[0].h)}</b> won the night, ${money(best[0].t.profit)} to ${money(best[1].t.profit)}`;

    } else if (nights.some(n => n.t.open)) verdict = 'Picks pending';

    const sides = nights.map(n => `<div class="side">

        <h4><span class="who"><span class="dot" style="background:${color(n.h)}"></span>${esc(n.h)}</span>

          <small class="${cls(n.t.profit)}">${n.t.graded + n.t.p ? `${rec(n.t)}, ${money(n.t.profit)}` : ''}</small></h4>

        ${n.ps.map(p => {

          const odds = p.boost != null && p.odds != null ? `<s>${signed(p.odds)}</s>${signed(p.boost)}` : p.eff != null ? signed(p.eff) : '';

          return `<div class="pick"><span class="seg">${esc(p.seg.label)}</span><span class="bet">${esc(p.text)}</span>

            <span class="right"><span class="odds">${odds}</span><span class="res ${p.result}">${LABEL[p.result]}</span></span></div>`;

        }).join('')}

      </div>`).join('');

    const d = s.info.date.getTime() ? s.info.date.toLocaleDateString('en-US', { weekday: 'short', month: 'short', day: 'numeric' }) : '';

    return `<article class="show"><div class="show-head"><h3>${esc(s.info.title)}</h3><p>${esc(d)}${verdict ? (d ? '. ' : '') + verdict : ''}</p></div><div class="sides">${sides}</div></article>`;

  }).join('');

}


/* ---------- load ---------- */

async function load() {

  $('updated').textContent = 'Loading…';

  $('notice').innerHTML = '';

  try {

    const url = PUBLISHED_CSV_URL || `https://docs.google.com/spreadsheets/d/${SHEET_ID}/gviz/tq?tqx=out:csv&headers=1&sheet=${encodeURIComponent(TAB)}`;

    const res = await fetch(url, { cache: 'no-store' });

    if (!res.ok) throw new Error(`The sheet answered with error ${res.status}.`);

    const text = await res.text();

    if (/^\s*</.test(text)) throw new Error('The sheet sent back a sign-in page instead of data.');

    Object.assign(state, build(toObjects(parseCSV(text))));

    render();

    $('updated').textContent = 'Updated ' + new Date().toLocaleTimeString([], { hour: 'numeric', minute: '2-digit' });

  } catch (e) {

    $('updated').textContent = 'Not loaded';

    $('notice').innerHTML = `<p class="note err">The picks didn’t load. ${esc(e.message)} Check that the responses sheet is shared as “Anyone with the link: Viewer.”</p>`;

  }

}
